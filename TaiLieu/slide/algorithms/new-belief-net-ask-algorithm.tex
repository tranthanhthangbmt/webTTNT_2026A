\code{
\func{Belief-Net-Ask}{X{\ac}E}{$\pv(X|E)$}
    \firstinputs{X}{a random variable}
    \inputs{E}{a set of evidence variables}
\bodysep
    \prog{Support-Except}(\var{X{\ac}null})
\fnsep
\func{Support-Except}{\var{X{\ac}V}}{$\pv(X|E_{X\setminus V})$}
\bodysep
    \key{if} $\var{X} \elt \var{E}$ \key{then return} observed point distribution for \var{X}
    \key{else} 
          calculate $\pv(E_{X\setminus V}^-|X)$ = \prog{Evidence-Except}(\var{X}{\ac}\var{V})
          \setq{$\mbf{U}$}{\noprog{Parents}[\var{X}]}
          \key{if} $\mbf{U}$ is empty 
                \key{then return}\ \ $\alpha \pv(E_{X\setminus V}^-|X) \pv(X)$
          \key{else}
                \key{for each} $U_i$ \key{in} \var{{\mbf U}}
                      calculate and store $\pv(U_i|E_{U_{i} \setminus X})$ = \prog{Support-Except}($U_i${\ac}\var{X})
                \key{return}\ \ $\alpha \pv(E_{X\setminus V}^-|X) \sum\limits_{\smbf{u}} \pv(X|\mbf{u}) \prod\limits_i \pv(U_i|E_{u_i\setminus X})$
\subfnsep
\fnsep
\func{Evidence-Except}{\var{X{\ac}V}}{$\pv(E^-_{X\setminus V}|X)$}
\bodysep
    \setq{$\mbf{Y}$}{$\noprog{Children}[\var{X}] - \var{V}$}
    \key{if} $\mbf{Y}$ is empty 
          \key{then return} a vector of 1s
    \key{else}
          \key{for each} $Y_i$ \key{in} $\mbf{Y} - \var{E}$ \key{do}NOOO!!!
                calculate $\pv(E_{Y_i}^-|y_i)$ = \prog{Evidence-Except}($Y_i${\ac}null)
                \setq{$\mbf{Z}_i$}{$\noprog{Parents}[Y_i] - \var{X}$}
                \key{for each} $Z_{ij}$ \key{in} $\mbf{Z}_i - \var{E}$
                      calculate $\pv(Z_{ij}|E_{Z_{ij}\setminus Y_i})$ = \prog{Support-Except}($Z_{ij}{\ac}Y_i$)
          \key{return}\ \ $\beta \prod\limits_i \sum\limits_{y_i} P(E_{Y_i}^-|y_i) \sum\limits_{\smbf{z}_i} \pv(y_i|X{\ac}\mbf{z}_i) \prod\limits_j {P(z_{ij}|E_{Z_{ij}\setminus Y_i})}$
\subfnsep}
