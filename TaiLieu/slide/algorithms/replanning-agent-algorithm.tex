\code{
\func{Replanning-Agent}{{\ts}\var{percept}}{an \var{action}}
    \firststatic{KB}{a knowledge base (includes action descriptions)}
    \static{p}{an annotated plan, initially $NoPlan$}
    \static{q}{an annotated plan, initially $NoPlan$}
    \static{G}{a goal}
\bodysep
    \noprog{Tell}(\var{KB}{\ac}\prog{Make-Percept-Sentence}({\ts}\var{percept{\ac}t}))
    \setq{\var{current}}{\prog{State-Description}(\var{KB{\ac}t})}
    \key{if} $\var{p} = NoPlan$ \key{then} 
          \setq{\var{p}}{\prog{Planner}(\var{current{\ac}G{\ac}KB})}
          \setq{\var{q}}{\var{p}}
          \key{if} $\var{p} = NoPlan$ \key{or} \var{p} is empty \key{then return} $NoOp$
    \key{if} \prog{Preconditions}({\ts}\var{p}) not currently true in \var{KB} \key{then}
          \setq{$p'$}{\prog{Choose-Best-Continuation}(\var{current{\ac}q})}
          \setq{\var{p}}{\prog{Append}(\prog{Planner}(\var{current}{\ac}\prog{Preconditions}({\ts}$p'$){\ac}\var{KB}){\ac}$p'$)}
          \setq{\var{q}}{\var{p}}
    \setq{\var{action}}{\prog{First}({\ts}\var{p})}
    \setq{\var{p}}{\prog{Rest}({\ts}\var{p})}
    \key{return} \var{action}
}
