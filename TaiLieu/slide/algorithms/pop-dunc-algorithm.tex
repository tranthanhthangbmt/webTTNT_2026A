\code{
\func{Select-Sub-Goal}{{\ts}\var{plan}}{\var{plan{\ac}precondition conjunct}}
\bodysep
    pick a plan step $S_{need}$ from \noprog{Steps}({\ts}\var{plan}) with a precondition conjunct $c$ that has not been achieved
    \key{if} $c$ is a universally quantified expression \key{then}
          \key{return} $S_{need}$, \prog{Expansion}($c$)
    \key{else if} $c$ is a disjunction $c_1 \lor c_2$ \key{then}
          \key{return} $S_{need}$, \key{choose}($c_1,c_2$)
    \key{else return} $S_{need}, c$
\fnsep
\proc{Choose-Operator}{\var{plan{\ac}operators{\ac}$S_{need}${\ac}c}}
\bodysep
    \key{choose} a step $S_{add}$ from \var{operators} or \noprog{Steps}({\ts}\var{plan}) that has $c_{add}$ as an effect
    \phantom{\key{choose}} such that \var{u} $=$ \prog{Unify}(c{\ac}$c_{add}${\ac}\noprog{Bindings}({\ts}\var{plan}))
    \key{if} there is no such step \key{then fail}
    \setq{$u'$}{\var{u} without the universally quantified variables of $c_{add}$}
    add $u'$ to \noprog{Bindings}({\ts}\var{plan})
    add $\cl{S_{add}}{c}{S_{need}}$ to \noprog{Links}({\ts}\var{plan})
    add $S_{add} \before S_{need}$ to \noprog{Orderings}({\ts}\var{plan})
    \key{if} $S_{add}$ is a newly added step from \var{operators} \key{then}
          add $S_{add}$ to \noprog{Steps}({\ts}\var{plan})
          add $Start \before S_{add} \before Finish$ to \noprog{Orderings}({\ts}\var{plan})
\fnsep
\proc{Resolve-Threats}{\var{plan}}
\bodysep
    \key{for each} $\cl{S_i}{c}{S_j}$ \key{in} \noprog{Links}({\ts}\var{plan}) \key{do}
          \key{for each} $S_{threat}$ \key{in} \noprog{Steps}({\ts}\var{plan}) \key{do}
                \key{for each} $c'$ \key{in} \prog{Effect}($S_{threat}$) \key{do}
                      \key{if} $\noprog{Subst}(\noprog{Bindings}({\ts}\var{plan}){\ac}c) = \noprog{Subst}(\noprog{Bindings}({\ts}\var{plan}){\ac}\lnot c')$ \key{then} 
                            \key{choose} either
                                  {\it Promotion:} Add $S_{threat}\before S_i$ to \noprog{Orderings}({\ts}\var{plan}) 
                                  {\it Demotion:} Add $S_j \before S_{threat}$ to \noprog{Orderings}({\ts}\var{plan}) 
                                  {\it Confrontation}: \key{if} $c'$ is really of the form $(c' \key{when} p)$ \key{then}
                                        \prog{Choose-Operator}({\ts}\var{plan{\ac}operators}{\ac}$S_{threat}, \lnot p$)
                                        \prog{Resolve-Threats}({\ts}\var{plan}) 
                            \key{if not} \prog{Consistent}({\ts}\var{plan}) \key{then fail}
                \key{end}
          \key{end}
    \key{end}
}
