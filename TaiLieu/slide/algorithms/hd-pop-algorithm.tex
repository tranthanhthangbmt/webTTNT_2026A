\code{
\func{HD-POP}{{\ts}\var{plan{\ac}operators{\ac}methods}}{\var{plan}}
    \firstinputs{plan}{an abstract plan with start and goal steps (and possibly other steps)}
\bodysep
    \key{loop do}
          \key{if} \prog{Solution?}({\ts}\var{plan}) \key{then return} \var{plan}
          \setq{$S_{need}, c$}{\prog{Select-Sub-Goal}({\ts}\var{plan})}
          \prog{Choose-Operator}({\ts}\var{plan{\ac}operators}{\ac}$S_{need}${\ac}\var{c})
          \setq{$S_{nonprim}$}{\prog{Select-Nonprimitive}({\ts}\var{plan})}
          \prog{Choose-Decomposition}({\ts}\var{plan{\ac}methods}{\ac}$S_{nonprim}$)
          \prog{Resolve-Threats}({\ts}\var{plan})
    \key{end}
}
