\code{
\func{Genetic-Algorithm}{{\ts}\var{population{\ac}\fnvar{Fitness-Fn}}}{an individual}
    \firstinputs{population}{a set of individuals} 
    \inputs{\fnvar{Fitness-Fn}}{a function that measures the fitness of an individual} 
\bodysep
    \key{repeat} 
          \setq{\var{parents}}{\prog{Selection}({\ts}\var{population}{\ac}\fnvar{Fitness-Fn})}
          \setq{\var{population}}{\prog{Reproduction}({\ts}\var{parents})} 
    \key{until} some individual is fit enough
    \key{return} the best individual in \var{population}, according to \fnvar{Fitness-Fn}
}
