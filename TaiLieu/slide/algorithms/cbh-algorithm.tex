\code{
\func{Current-Best-Learning}{\var{examples}}{a hypothesis}
\bodysep
    \setq{\var{H}}{any hypothesis consistent with the first example in \var{examples}}
    \key{for} \key{each} remaining example in \var{examples} \key{do}
          \key{if} \var{e} is false positive for \var{H} \key{then}
                \setq{\var{H}}{\key{choose} a specialization of \var{H} consistent with \var{examples}}
          \key{else} \key{if} \var{e} is false negative for \var{H} \key{then}
                \setq{\var{H}}{\key{choose} a generalization of \var{H} consistent with \var{examples}}
          \key{if} no consistent specialization/generalization can be found \key{then fail} 
    \key{end}
    \key{return} \var{H}
}
