\code{
\func{CPOP}{\var{initial{\ac}goals{\ac}operators}}{\var{plan}}
\bodysep
    \setq{\var{plan}}{\prog{Make-Plan}(\var{initial{\ac}goals})} 
    \key{loop do}
\subfnsep
          {\it Termination}:
                \key{if} there are no unsatisfied preconditions
                \phantom{\key{if}}  and the contexts of the finish steps are exhaustive 
                      \key{then return} \var{plan}
\subfnsep
          {\it Alternative context generation}:
                \key{if} the plans for existing finish steps are complete and have contexts $C_1 \ldots C_n$ \key{then}
                      add a new finish step with a context $\lnot (C_1\lor \ldots \lor C_n)$
                      this becomes the {\it current context}
\subfnsep
          {\it Subgoal selection and addition}:
                find a plan step $S_{need}$ with an open precondition $c$
\subfnsep
          {\it Action selection}:
                \key{choose} a step $S_{add}$ from \var{operators} or \noprog{Steps}({\ts}\var{plan}) that adds $c$ or
                \phantom{\key{choose}} knowledge of $c$ and has a context compatible with the current context
                \key{if} there is no such step 
                      \key{then fail}
                add $\cl{S_{add}}{c}{S_{need}}$ to \noprog{Links}({\ts}\var{plan})
                add $S_{add} < S_{need}$ to \noprog{Orderings}({\ts}\var{plan})
                \key{if} $S_{add}$ is a newly added step \key{then}
                      add $S_{add}$ to \noprog{Steps}({\ts}\var{plan})
                      add $Start < S_{add} < Finish$ to \noprog{Orderings}({\ts}\var{plan})
\subfnsep
          {\it Threat resolution}:
                \key{for each} step $S_{threat}$ that potentially threatens any causal link $\cl{S_i}{c}{S_j}$
                \phantom{\key{for each}} with a compatible context \key{do} 
                      \key{choose} one of
                            {\it Promotion:} Add $S_{threat}<S_i$ to \noprog{Orderings}({\ts}\var{plan})
                            {\it Demotion:} Add $S_j < S_{threat}$ to \noprog{Orderings}({\ts}\var{plan})
                            {\it Conditioning}: 
                                  find a conditional step $S_{cond}$ possibly before both $S_{threat}$ and $S_j$, where 
                                        1. the context of $S_{cond}$ is compatible with the contexts of $S_{threat}$ and $S_j$;
                                        2. the step has outcomes consistent with $S_{threat}$ and $S_j$, respectively 
                                  add conditioning links for the outcomes from $S_{cond}$ to $S_{threat}$ and $S_j$
                                  augment and propagate the contexts of $S_{threat}$ and $S_j$
\subfnsep
                      \key{if} no choice is consistent 
                                  \key{then fail}
                \key{end}
    \key{end}
}
