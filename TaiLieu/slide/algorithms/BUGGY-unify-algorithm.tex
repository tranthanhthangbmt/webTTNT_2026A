\code{
\func{Unify}{\var{x{\ac}y}}{a substitution to make \var{x} and \var{y} identical, if possible}
    \firstinputs{x}{a variable, constant, or compound}
    \inputs{y}{a variable, constant, or compound}
\bodysep
    \key{if} \var{x} = \var{y} \key{then} \key{return} \{\}
    \key{else if} \prog{Variable?}(\var{x}) \key{then return} \prog{Unify-Var}(\var{x}{\ac}\var{y})
    \key{else if} \prog{Variable?}(\var{y}) \key{then return} \prog{Unify-Var}(\var{y}{\ac}\var{x})
    \key{else if} \prog{Compound?}(\var{x}) \key{and} \prog{Compound?}(\var{y}) \key{and} \prog{Op}[\var{x}] = \prog{Op}[\var{y}] \key{then}
          \key{return} \prog{Unify-Lists}(\{\}{\ac}\prog{Args}(\var{x}){\ac}\prog{Args}(\var{y}))
    \key{else return} failure
\fnsep
\func{Unify-Var}{\var{var{\ac}x}}{a substitution}
    \firstinputs{var}{a variable}
    \inputs{x}{any compound}
\bodysep
    \prog{Occur-Check}:
        \key{if} \var{var} occurs anywhere in \var{x} 
              \key{then} \key{return} failure 
        \key{else return} \{\var{x}/\var{var}\}
\fnsep
\func{Unify-Lists}{\var{$\theta${\ac}x-args{\ac}y-args}}{$\theta$ or failure}
    \firstinputs{$\theta$}{the ``substitution so far''}
    \inputs{x-args}{list of expressions}
    \inputs{y-args}{list of expressions}
\bodysep
    \key{if} $\theta$ = Fail \key{then return} failure
    \key{else if} \var{x-args} = [$\,$] \key{and} \var{y-args} = [$\,$] \key{then return} $\theta$
    \key{else if} \var{x-args} = [$\,$] \key{or} \var{y-args} = [$\,$] \key{then return} failure
    \key{else} 
          \setq{\var{$\theta_2$}}{\prog{Unify}(\noprog{Subst}($\theta${\ac}\prog{First}(\var{x-args})){\ac}\noprog{Subst}($\theta${\ac}\prog{First}(\var{y-args})))}
          \setq{\var{$\theta_3$}}{\prog{Unify-Lists}(\var{$\theta_2$}{\ac}\prog{Rest}(\var{x-args}){\ac}\prog{Rest}(\var{y-args}))}
          \key{return} \prog{Compose}(\var{$\theta_2$}{\ac}\var{$\theta_3$})
}
