\code{
\func{Run-Eval-Environment}{\var{state{\ac}\fnvar{Update-Fn}{\ac}agents{\ac}\\\phantom{\key{function} \prog{Run-Eval-Environment}(}termination{\ac}\fnvar{Performance-Fn}}}{\var{scores}}
    \firstlocal{scores}{a vector the same size as \var{agents}, all 0}
\bodysep
    \key{repeat}
          \key{for each} \var{agent} \key{in} \var{agents} \key{do}
                \setq{\prog{Percept}[\var{agent}]}{\prog{Get-Percept}(\var{agent}{\ac}\var{state})}
          \key{end}
          \key{for each} \var{agent} \key{in} \var{agents} \key{do}
                \setq{\prog{Action}[\var{agent}]}{\prog{Program}[\var{agent}](\prog{Percept}[\var{agent}])}
          \key{end}
          \setq{\var{state}}{\fnvar{Update-Fn}(\var{actions{\ac}agents{\ac}state})}
          \setq{\var{scores}}{\fnvar{Performance-Fn}(\var{scores{\ac}agents{\ac}state})} 
    \key{until} \var{termination}(\var{state})
    \key{return} \var{scores} \rcmt{change}
}
