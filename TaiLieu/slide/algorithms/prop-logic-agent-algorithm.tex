\code{
\func{Propositional-KB-Agent}{{\ts}\var{percept}}{an \var{action}}
    \firststatic{KB}{a knowledge base}
    \static{t}{a counter, initially 0, indicating time}
\bodysep
    \noprog{Tell}(\var{KB}{\ac}\prog{Make-Percept-Sentence}({\ts}\var{percept{\ac}t}))
    \key{for each} \var{action} \key{in} the list of possible actions \key{do}
          \key{if} {\noprog{Ask}(\var{KB}{\ac}\prog{Make-Action-Query}(\var{t{\ac}action}))} \key{then}
                \setq{\var{t}}{\var{t} + 1}
                \key{return} \var{action}
    \key{end}
