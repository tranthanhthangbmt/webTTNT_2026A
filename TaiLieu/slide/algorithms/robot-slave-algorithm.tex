\code{
\func{Simple-Communicating-Agent}{{\ts}\var{percept}}{\var{action}}
    \firststatic{KB}{a knowledge base}
    \static{t}{a counter, initially 0, indicating time}
\bodysep
    \noprog{Tell}(\var{KB}{\ac}\prog{Make-Percept-Sentence}({\ts}\var{percept{\ac}t}))
    \setq{\var{words}}{\prog{Speech-Part}({\ts}\var{percept})}
    \setq{\var{semantics}}{\prog{Disambiguate}(\prog{Semantics}(\prog{Parse}(\var{words})))}
    \key{if} \prog{Type}[\var{semantics}] = $Command$ \key{then}
          \setq{action}{\prog{Contents}[\var{semantics}]}
    \key{else if} \prog{Type}[\var{semantics}] = $Statement$ \key{then}
          \noprog{Tell}(\var{KB}{\ac}\prog{Contents}[\var{semantics}])
          \setq{action}{\noprog{Ask}(\var{KB}{\ac}\prog{Make-Action-Query}({\ts}\var{percept{\ac}t}))}
    \key{else if} \prog{Type}[\var{semantics}] = $None$ \key{then}
          \setq{action}{\noprog{Ask}(\var{KB}{\ac}\prog{Make-Action-Query}({\ts}\var{percept{\ac}t}))}
    \setq{description}\prog{Generate-Description}({\ts}\var{percept})
    \key{return} \prog{Compound-Action}(\prog{Say}(\var{description}){\ac}\prog{Do}(\var{action}))
}
