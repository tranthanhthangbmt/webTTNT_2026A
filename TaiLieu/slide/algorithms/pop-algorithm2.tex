\code{
\proc{Choose-Operator}{\var{plan{\ac}operators{\ac}$S_{need}${\ac}c}}
\bodysep
    \key{choose} chọn một bước $S_{add}$ từ \var{operators} hoặc \noprog{Steps}({\ts}\var{plan}) có hiệu ứng là $c$
    \key{if} nếu không có bước như vậy \key{then fail} 
    thêm liên kết nhân quả $\cl{S_{add}}{c}{S_{need}}$ vào \noprog{Links}({\ts}\var{plan})
    thêm ràng buộc thứ tự $S_{add} \before S_{need}$ vào \noprog{Orderings}({\ts}\var{plan})
    \key{if} nếu $S_{add}$ là một bước mới được thêm từ \var{operators} \key{then}
          thêm $S_{add}$ vào \noprog{Steps}({\ts}\var{plan})
          thêm $Start \before S_{add} \before Finish$ vào \noprog{Orderings}({\ts}\var{plan})
\fnsep
\proc{Resolve-Threats}{\var{plan}}
\bodysep
    \key{for each} với mỗi $S_{threat}$ đe dọa liên kết $\cl{S_i}{c}{S_j}$ trong \noprog{Links}({\ts}\var{plan}) \key{do}
          \key{choose} chọn hoặc
                {\it Hạ hạng (Demotion):} Thêm $S_{threat}\before S_i$ vào \noprog{Orderings}({\ts}\var{plan})
                {\it Thăng hạng (Promotion):} Thêm $S_j \before S_{threat}$ vào \noprog{Orderings}({\ts}\var{plan})
          \key{if not} nếu không \prog{Consistent}({\ts}\var{plan}) \key{then fail}
    \key{end}
}
