\code{
\nofunc{SMA*}{\var{problem}}{a solution sequence}
    \firstinputs{problem}{a problem}
    \firstlocal{Queue}{a queue of nodes ordered by $f$-cost}
\bodysep
    \setq{\var{Queue}}{\prog{Make-Queue}(\prog{Make-Node}(\prog{Initial-State}[\var{problem}]))}
    \key{loop do}
          \key{if} \var{Queue} is empty \key{then return} failure
          \setq{\var{n}}{deepest least-f-cost node in \var{Queue}}
          \key{if} \prog{Goal-Test}(\var{n}) \key{then return} success
          \setq{\var{s}}{\prog{Next-Successor}(\var{n})}
          \key{if} \var{s} is not a goal and is at maximum depth \key{then}
                \setq{f(\var{s})}{$\infty$}
          \key{else} 
                \setq{f(\var{s})}{\prog{Max}(f(\var{n}){\ac}g(\var{s})+h(\var{s}))}
          \key{if} all of \var{n}'s successors have been generated \key{then}
                update \var{n}'s $f$-cost and those of its ancestors if necessary
          \key{if} \prog{Successors}(\var{n}) all in memory \key{then} remove \var{n} from \var{Queue} 
          \key{if} memory is full \key{then}
                delete shallowest, highest-f-cost node in \var{Queue}
                remove it from its parent's successor list	
                insert its parent on \var{Queue} if necessary
          insert \var{s} on \var{Queue}
    \key{end}
}
