\code{
\func{Decision-List-Learning}{\var{examples}}{a decision list, \var{No} or failure}
\bodysep
    \key{if} \var{examples} is empty \key{then return} the value \var{No}  
    \setq{t}{a test that matches a nonempty subset $\var{examples}_t$ of \var{examples}}
    \phantom{\setq{t}{}} such that the members of $\var{examples}_t$ are all positive or all negative
    \key{if} there is no such \var{t} \key{then return} failure
    \key{if} the examples in $\var{examples}_t$ are positive \key{then} \setq{\var{o}}{\var{Yes}}
    \key{else} \setq{\var{o}}{\var{No}}
    \key{return} a decision list with initial test \var{t} and outcome \var{o}
    \phantom{\key{return}} and remaining elements given by \prog{Decision-List-Learning}($\var{examples} - \var{examples}_t$)
}
