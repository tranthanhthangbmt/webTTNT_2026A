\code{
\func{IDA*}{{\ts}\var{problem}}{a solution sequence}
    \firstinputs{problem}{a problem}
    \firstlocal{f-limit}{the current $f$-{\sc Cost} limit}
    \local{root}{a node}
\bodysep
    \setq{\var{root}}{\prog{Make-Node}(\prog{Initial-State}[\var{problem}])}
    \setq{\var{f-limit}}{$f$-{\sc Cost}[\var{root}]}\index{f-Cost@$f$-{\sc Cost}}
    \key{loop do}
          \setq{\var{solution{\ac}f-limit}}{\prog{DFS-Contour}(\var{root{\ac}f-limit})}
          \key{if} \var{solution} is non-null \key{then return} \var{solution}
          \key{if} \var{f-limit} = $\infty$ \key{then return} failure; \key{end}
\fnsep
\func{DFS-Contour}{\var{node{\ac}f-limit}}{a solution sequence and a new $f$-{\sc Cost} limit}
    \firstinputs{node}{a node}
    \inputs{f-limit}{the current $f$-{\sc Cost} limit}
    \firstlocal{next-f}{the $f$-{\sc Cost} limit for the next contour, initially $\infty$}
\bodysep
    \key{if} {$f$-{\sc Cost}}[\var{node}] $>$ \var{f-limit} \key{then return} null, {$f$-{\sc Cost}}[\var{node}]
    \key{if} \prog{Goal-Test}[\var{problem}](\noprog{State}[\var{node}]) \key{then return} \var{node}, \var{f-limit}
    \key{for each} node \var{s} \key{in} \prog{Successors}(\var{node}) \key{do}
          \setq{\var{solution}{\ac}\var{new-f}}{\prog{DFS-Contour}(\var{s{\ac}f-limit})}
          \key{if} \var{solution} is non-null \key{then return} \var{solution{\ac}f-limit} 
          \setq{\var{next-f}}{\prog{Min}(\var{next-f{\ac}new-f})}; \key{end}
    \key{return} null, \var{next-f} 
}
