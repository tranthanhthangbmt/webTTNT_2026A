\code{
\proc{Run-Environment}{\var{state{\ac}\fnvar{Update-Fn}{\ac}agents{\ac}termination}}
    \firstinputs{state}{the initial state of the environment}
    \inputs{\fnvar{Update-Fn}}{function to modify the environment}
    \inputs{agents}{a set of agents}
    \inputs{termination}{a predicate to test when we are done}
\bodysep
    \key{repeat}
          \key{for each} \var{agent} \key{in} \var{agents} \key{do}
                \setq{\prog{Percept}[\var{agent}]}{\prog{Get-Percept}(\var{agent}{\ac}\var{state})}
          \key{end}
          \key{for each} \var{agent} \key{in} \var{agents} \key{do}
                \setq{\prog{Action}[\var{agent}]}{\prog{Program}[\var{agent}](\prog{Percept}[\var{agent}])}
          \key{end}
          \setq{\var{state}}{\fnvar{Update-Fn}(\var{actions{\ac}agents{\ac}state})}
    \key{until} \var{termination}(\var{state})
}
